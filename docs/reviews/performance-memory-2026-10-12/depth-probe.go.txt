package evaly

import (
 "context"
 "encoding/json"
 "fmt"
 "strings"
 "testing"
)

func TestDepthProbe(t *testing.T) {
 for _, depth := range []int{9998,9999,10000} {
  for _, mode:=range []string{"input","projected-fresh","projected-saturated"} {
   t.Run(fmt.Sprintf("%s/%d",mode,depth),func(t *testing.T){
    payload:=json.RawMessage(strings.Repeat("[",depth)+"0"+strings.Repeat("]",depth))
    p:=&captureMemoryPolicy{project:func(e Event)(Event,error){return e,nil}}
    c,_:=NewCapture(CaptureConfig{Policy:p,KnownKinds:[]string{"output"},MaxEvents:1,MaxBytes:1<<20})
    e:=Event{Version:1,Sequence:1,Kind:"output",Payload:json.RawMessage(`0`)}
    if mode=="projected-saturated" {if err:=c.Record(context.Background(),e);err!=nil {t.Fatal(err)};e.Sequence=2}
    if mode=="input" {e.Payload=payload} else {p.project=func(e Event)(Event,error){e.Payload=payload;return e,nil}}
    p.calls=0
    err:=c.Record(context.Background(),e)
    t.Logf("Record err=%v policy calls=%d retained=%d",err,p.calls,len(c.record.Events))
   })
  }
 }
}

func TestInvalidStringSizeProbe(t *testing.T) {
 e:=Event{Version:1,Sequence:1,Kind:"output",CorrelationID:string([]byte{0xff,0xfe}),Payload:json.RawMessage(`0`)}
 b,err:=canonical(e);if err!=nil {t.Fatal(err)}
 p:=&captureMemoryPolicy{project:func(raw Event)(Event,error){raw.CorrelationID=e.CorrelationID;return raw,nil}}
 c,_:=NewCapture(CaptureConfig{Policy:p,KnownKinds:[]string{"output"},MaxEvents:1,MaxBytes:len(b)})
 err=c.Record(context.Background(),e)
 t.Logf("Record err=%v countedbytes=%d want=%d",err,c.bytes,len(b))
}
